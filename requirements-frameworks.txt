# ============================================================================
# 第 4 章《主流 Agent 框架》对比演示 —— 可选依赖
#
# 适用环境：与根目录 requirements.txt 同一个 .venv（Python 3.12）
# 安装：
#   .venv/Scripts/python.exe -m pip install -r requirements-frameworks.txt
#
# 【为什么不放进 requirements.txt】
#   这四组演示依赖各自成体系、体积大，且与主栈的版本偏好不同（例如 langgraph
#   只有预发布版与旧代码匹配）。单独一份清单可以让主环境的版本组合保持已验证状态，
#   出现冲突时也只影响本文件，不会牵连第 1–3、5、6 章。
#
# 【版本选择依据】
#   多数版本取自本机原有 Python 3.10 环境 —— 那套组合已实际跑通过第 4 章四个演示，
#   因此沿用同样版本，避免版本漂移带来的 API 差异。两处例外（均为解决冲突）：
#   camel-ai 0.2.75 → 0.2.90、新增 mcp==1.29.1，理由见下面对应条目。
#
# 对应目录：
#   agentscope  → 4-agent-frame/AgentScopeDemo/
#   autogen     → 4-agent-frame/AutoGenDemo/
#   camel-ai    → 4-agent-frame/CAMEL/
#   langgraph   → 4-agent-frame/Langgraph/
# ============================================================================


# ---------------------------------------------------------------------------
# 0. 公共依赖修正：mcp 必须锁 1.x
# ---------------------------------------------------------------------------
# agentscope 与 camel-ai 都依赖 mcp，但两者都只写了下限（agentscope: >=1.13，
# camel-ai: >=1.3.0），没有上限 —— pip 会直接装到最新的 2.x，结果：
#   ImportError: cannot import name 'streamablehttp_client' from 'mcp.client'
# 因为 agentscope 1.0.2 内部仍在使用 mcp.client.streamable_http 的旧 API，
# 该接口在 mcp 2.x 已被移除。这里显式锁回旧环境实测可用的 1.29.1。
# 这条必须放在本文件最前，否则单独安装 agentscope 时同样的坑会复现。
mcp==1.29.1


# ---------------------------------------------------------------------------
# 1. AgentScope 演示（AgentScopeDemo/main_cn.py 等）
# ---------------------------------------------------------------------------
# 1.0.2 是旧环境实测可用的版本；PyPI 最新为 2.x（后续大版本，API 有变动且未验证），
# 因此锁定 1.0.2 与示例代码保持一致。
agentscope==1.0.2


# ---------------------------------------------------------------------------
# 2. AutoGen 演示（AutoGenDemo/autogen_software_team.py）
# ---------------------------------------------------------------------------
autogen-agentchat==0.7.5
autogen-core==0.7.5
# [openai] extra 提供 autogen_ext.models.openai.OpenAIChatCompletionClient
autogen-ext[openai]==0.7.5
# AutoGenDemo/output.py 用它做流式输出界面
streamlit==1.62.0


# ---------------------------------------------------------------------------
# 3. CAMEL 演示（CAMEL/DigitalBookWriting.py）
# ---------------------------------------------------------------------------
# 为什么不是旧环境里的 0.2.75：0.2.75 声明 pillow<11.0.0，而 autogen-core 0.7.5
# 要求 pillow>=11.0.0 —— 同装一个环境会直接 ResolutionImpossible。
# （旧 Python 3.10 环境里两者"共存"是因为分两次 pip 安装、未做一致性校验，
#   实际 pillow 停留在 10.4.0，autogen 的约束并未被满足。）
# 0.2.90 已放宽为 pillow>=10.0.0，冲突消除，且 0.2.75→0.2.90 为同系列小版本，
# 示例用到的 RolePlaying / ModelFactory / ModelPlatformType / print_text_animated
# 均未变动。
camel-ai==0.2.90
# 终端彩色输出
colorama==0.4.6


# ---------------------------------------------------------------------------
# 4. LangGraph 演示（Langgraph/Dialogue_System.py）
# ---------------------------------------------------------------------------
# langgraph 1.0.0a3 是预发布版；旧环境即用此版，LangGraph 1.0 正式版 API 有变动，
# 若升级需同步改示例代码。
langgraph==1.0.0a3
langchain-core==0.3.86
langchain-openai==0.3.33
